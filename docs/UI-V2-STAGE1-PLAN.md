# UI V2 — Stage 1 (Render Foundation) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the resolution-independent UI render foundation (`UiRenderer` + `UiText` + backends) defined in [UI-V2.md](UI-V2.md), promoting the validated PoC into a clean, durable architecture — with NO changes to the live client UI.

**Architecture:** A backend-swappable render layer in `com.club.ui`. Components (later stages) call only `Ui.renderer()` / `Ui.text()`. `ModernBackend` does analytic SDF shapes + MSDF text via custom core shaders; `LegacyBackend` is a fallback over `DrawContext`. Pure logic (color math, MSDF metrics, text layout, charset) is separated from Minecraft/GL so it is unit-testable; rendering is verified by a headless acceptance screen.

**Tech Stack:** Fabric 1.21.1, Yarn, Java 21, LWJGL3 GL, MSDF (msdf-atlas-gen v1.4), GLSL 150 core shaders, Gson, JUnit 5 (pure-logic tests).

## Global Constraints

- Do NOT modify `gui/*`, `hud/*`, `Theme`, `ClubFont`, `RenderHelper`, `ClubClient`, `ClubMod`, existing mixins or `gui/sandbox/*`. New code lives only in `com.club.ui.*` and `assets/club/ui/*` (+ one additive client entrypoint).
- **Hard rule:** outside `com.club.ui.backend.*`, NO direct `DrawContext.fill`, `DrawContext.drawText`, `RenderSystem.*`, `BufferBuilder`/`Tessellator`/`BufferRenderer`, `GlUniform`, `ShaderProgram`, `ClubFont`, `RenderHelper`. All rendering goes through `Ui.renderer()` / `Ui.text()`.
- **Charset (frozen once):** Latin + Cyrillic + Punctuation + Basic symbols. Generated for Inter Regular/Medium/SemiBold and never regenerated casually.
- ModernBackend forbids: vanilla `TextRenderer` as primary text, CPU-AA, per-pixel `fill`, glow-from-quads, banding gradients.
- No literals of colour/size inside future components (enforced from Stage 2; Stage 1 only builds the renderer + token-free acceptance screen).
- Coordinates: float, GUI-scaled px, origin top-left. Colour: packed ARGB `int`.
- Java 21, `options.release = 21`. Commit after every task.

package com.club.ui;

/** Pure ARGB (0xAARRGGBB) helpers. No Minecraft/GL dependencies. */
public final class Color {
    private Color() {}
    public static int rgba(int r, int g, int b, int a) {
        return ((a & 0xFF) << 24) | ((r & 0xFF) << 16) | ((g & 0xFF) << 8) | (b & 0xFF);
    }
    public static int withAlpha(int c, int a) { return (c & 0x00FFFFFF) | ((a & 0xFF) << 24); }
    public static int scaleAlpha(int c, float mul) {
        int a = Math.round(((c >>> 24) & 0xFF) * Math.max(0f, Math.min(1f, mul)));
        return (c & 0x00FFFFFF) | (a << 24);
    }
    public static int lerp(int x, int y, float t) {
        t = Math.max(0f, Math.min(1f, t));
        int xa = (x >>> 24) & 0xFF, xr = (x >>> 16) & 0xFF, xg = (x >>> 8) & 0xFF, xb = x & 0xFF;
        int ya = (y >>> 24) & 0xFF, yr = (y >>> 16) & 0xFF, yg = (y >>> 8) & 0xFF, yb = y & 0xFF;
        return rgba(Math.round(xr + (yr - xr) * t), Math.round(xg + (yg - xg) * t),
                    Math.round(xb + (yb - xb) * t), Math.round(xa + (ya - xa) * t));
    }
    public static float af(int c) { return ((c >>> 24) & 0xFF) / 255f; }
    public static float rf(int c) { return ((c >>> 16) & 0xFF) / 255f; }
    public static float gf(int c) { return ((c >>> 8) & 0xFF) / 255f; }
    public static float bf(int c) { return (c & 0xFF) / 255f; }
}

package com.club.ui.text;

/** Frozen codepoint coverage for the MSDF atlases: Latin + Latin-1 + Cyrillic + punctuation/symbols. */
public final class Charset {
    private Charset() {}
    public static boolean contains(int cp) {
        if (cp >= 0x20 && cp <= 0x7E) return true;        // ASCII
        if (cp >= 0xA0 && cp <= 0xFF) return true;        // Latin-1 supplement
        if (cp >= 0x0400 && cp <= 0x04FF) return true;    // Cyrillic
        switch (cp) {                                      // common punctuation/symbols
            case 0x2013: case 0x2014: case 0x2018: case 0x2019:
            case 0x201C: case 0x201D: case 0x2022: case 0x2026:
            case 0x2192: case 0x221E: return true;
        }
        return false;
    }
}

package com.club.ui.text;

import com.google.gson.JsonElement;
import com.google.gson.JsonObject;
import com.google.gson.JsonParser;
import java.util.HashMap;
import java.util.Map;

/** Pure model + parser of msdf-atlas-gen JSON. No Minecraft/GL dependencies. */
public final class MsdfMetrics {
    public static final class Glyph {
        public float advance;
        public boolean hasBounds;
        public float pl, pb, pr, pt;     // plane bounds, em (baseline-relative, y up)
        public float u0, v0, u1, v1;     // atlas UVs (top-left origin)
    }

    public int atlasW, atlasH;
    public float distanceRange;
    public float emLineHeight, emAscender, emDescender;
    private final Map<Integer, Glyph> glyphs = new HashMap<>();

    public static MsdfMetrics parse(String json) {
        MsdfMetrics m = new MsdfMetrics();
        JsonObject root = JsonParser.parseString(json).getAsJsonObject();
        JsonObject atlas = root.getAsJsonObject("atlas");
        m.atlasW = atlas.get("width").getAsInt();
        m.atlasH = atlas.get("height").getAsInt();
        m.distanceRange = atlas.get("distanceRange").getAsFloat();
        JsonObject me = root.getAsJsonObject("metrics");
        m.emLineHeight = me.get("lineHeight").getAsFloat();
        m.emAscender = me.get("ascender").getAsFloat();
        m.emDescender = me.get("descender").getAsFloat();
        for (JsonElement ge : root.getAsJsonArray("glyphs")) {
            JsonObject g = ge.getAsJsonObject();
            Glyph gl = new Glyph();
            gl.advance = g.get("advance").getAsFloat();
            if (g.has("planeBounds") && g.has("atlasBounds")) {
                JsonObject pb = g.getAsJsonObject("planeBounds");
                gl.pl = pb.get("left").getAsFloat();  gl.pr = pb.get("right").getAsFloat();
                gl.pb = pb.get("bottom").getAsFloat(); gl.pt = pb.get("top").getAsFloat();
                JsonObject ab = g.getAsJsonObject("atlasBounds");
                float al = ab.get("left").getAsFloat(), ar = ab.get("right").getAsFloat();
                float abm = ab.get("bottom").getAsFloat(), at = ab.get("top").getAsFloat();
                gl.u0 = al / m.atlasW; gl.u1 = ar / m.atlasW;
                gl.v0 = 1f - at / m.atlasH;   // atlas yOrigin=bottom → flip
                gl.v1 = 1f - abm / m.atlasH;
                gl.hasBounds = true;
            }
            m.glyphs.put(g.get("unicode").getAsInt(), gl);
        }
        return m;
    }

    public Glyph get(int cp) { return glyphs.get(cp); }
    public float advanceOf(int cp) { Glyph g = glyphs.get(cp); return g == null ? 0f : g.advance; }

    public float width(String s, float size) {
        float w = 0;
        for (int i = 0; i < s.length(); i++) {
            Glyph g = glyphs.get((int) s.charAt(i));
            if (g == null) g = glyphs.get((int) '?');
            if (g != null) w += g.advance * size;
        }
        return w;
    }
    public float lineHeight(float size) { return emLineHeight * size; }
    public float ascent(float size) { return emAscender * size; }
    public float descent(float size) { return -emDescender * size; }
}

    float aa = max(fwidth(d), 1e-4);

        float inner = 1.0 - smoothstep(-aa, aa, d + Thickness);

import com.club.ui.Axis;
import com.club.ui.Color;

import com.mojang.blaze3d.systems.RenderSystem;

import net.minecraft.client.render.VertexFormats;
import org.joml.Matrix4f;
import org.lwjgl.opengl.GL11;

import java.util.ArrayDeque;
import java.util.Deque;

/** UiRenderer over the analytic SDF shader. Resolution-independent. */
public final class ModernBackend implements UiRenderer {
    private DrawContext ctx;                       // set each frame by Ui/screen
    private final Deque<float[]> opacity = new ArrayDeque<>();
    private final Deque<int[]> scissor = new ArrayDeque<>();

    public void begin(DrawContext ctx) { this.ctx = ctx; opacity.clear(); }

    @Override public boolean isResolutionIndependent() { return true; }

    @Override public void rect(float x, float y, float w, float h, int c) {
        shape(x, y, w, h, 0f, Radii.all(0f), 0, c, c, 0, 0f, 0f);
    }
    @Override public void roundedRect(float x, float y, float w, float h, float r, int c) {
        shape(x, y, w, h, 0f, Radii.all(r), 0, c, c, 0, 0f, 0f);
    }
    @Override public void roundedRect(float x, float y, float w, float h, Radii r, int c) {
        shape(x, y, w, h, 0f, r, 0, c, c, 0, 0f, 0f);
    }
    @Override public void border(float x, float y, float w, float h, float r, float th, int c) {
        shape(x, y, w, h, 0f, Radii.all(r), 1, c, c, 0, 0f, th);
    }
    @Override public void gradient(float x, float y, float w, float h, float r, int a, int b, Axis axis) {
        shape(x, y, w, h, 0f, Radii.all(r), 3, a, b, axis == Axis.VERTICAL ? 1 : 0, 0f, 0f);
    }
    @Override public void glow(float x, float y, float w, float h, float r, float size, int c) {
        shape(x, y, w, h, size, Radii.all(r), 2, c, c, 0, size, 0f);
    }
    @Override public void shadow(float x, float y, float w, float h, float r, float dx, float dy, float blur, int c) {
        shape(x + dx, y + dy, w, h, blur, Radii.all(r), 2, c, c, 0, blur, 0f); // offset soft glow
    }
    @Override public void circle(float cx, float cy, float rr, int c) {
        roundedRect(cx - rr, cy - rr, rr * 2, rr * 2, rr, c);
    }
    @Override public void line(float x1, float y1, float x2, float y2, float th, int c) {
        // axis-aligned fast path; diagonal approximated by a thin rotated-less box around the midpoint span
        if (y1 == y2) { rect(Math.min(x1, x2), y1 - th / 2f, Math.abs(x2 - x1), th, c); return; }
        if (x1 == x2) { rect(x1 - th / 2f, Math.min(y1, y2), th, Math.abs(y2 - y1), c); return; }
        // general case: short segments (Stage 1 keeps lines axis-aligned in components)
        rect(Math.min(x1, x2), Math.min(y1, y2), Math.max(th, Math.abs(x2 - x1)), Math.max(th, Math.abs(y2 - y1)), c);
    }

    // --- clip (GL scissor stack; rounded approximated as rect in Stage 1) ---
    @Override public void pushClip(float x, float y, float w, float h) { pushScissor(x, y, w, h); }
    @Override public void pushRoundedClip(float x, float y, float w, float h, float r) { pushScissor(x, y, w, h); }
    @Override public void popClip() {
        if (!scissor.isEmpty()) scissor.pop();
        if (scissor.isEmpty()) RenderSystem.disableScissor();
        else { int[] s = scissor.peek(); RenderSystem.enableScissor(s[0], s[1], s[2], s[3]); }
    }
    private void pushScissor(float x, float y, float w, float h) {
        MinecraftClient mc = MinecraftClient.getInstance();
        double sc = mc.getWindow().getScaleFactor();
        int fbH = mc.getWindow().getFramebufferHeight();
        int sx = (int) Math.round(x * sc), sw = (int) Math.round(w * sc), sh = (int) Math.round(h * sc);
        int sy = fbH - (int) Math.round((y + h) * sc);   // GL scissor origin = bottom-left
        scissor.push(new int[]{sx, sy, sw, sh});
        RenderSystem.enableScissor(sx, sy, sw, sh);
    }

    // --- opacity stack ---
    @Override public void pushOpacity(float m) {
        float cur = opacity.isEmpty() ? 1f : opacity.peek()[0];
        opacity.push(new float[]{cur * Math.max(0f, Math.min(1f, m))});
    }
    @Override public void popOpacity() { if (!opacity.isEmpty()) opacity.pop(); }
    private float opacity() { return opacity.isEmpty() ? 1f : opacity.peek()[0]; }

    // --- core draw ---
    private void shape(float x, float y, float w, float h, float pad, Radii radii,
                       int mode, int colorA, int colorB, int gradAxis, float feather, float thickness) {
        if (!UiShaders.ready() || ctx == null || w <= 0 || h <= 0) return;
        ShaderProgram sh = UiShaders.SDF;
        float halfW = w / 2f, halfH = h / 2f, cx = x + halfW, cy = y + halfH;
        float x0 = x - pad, y0 = y - pad, x1 = x + w + pad, y1 = y + h + pad;
        int ca = Color.withAlpha(colorA, Math.round(((colorA >>> 24) & 0xFF) * opacity()));

        RenderSystem.enableBlend(); RenderSystem.defaultBlendFunc(); RenderSystem.disableCull();
        RenderSystem.setShader(() -> sh);
        u2(sh, "HalfSize", halfW, halfH);
        u4f(sh, "CornerRadii", radii.tl(), radii.tr(), radii.br(), radii.bl());
        ui(sh, "Mode", mode); ui(sh, "GradAxis", gradAxis);
        u1(sh, "Feather", Math.max(feather, 1e-4f)); u1(sh, "Thickness", Math.max(thickness, 1e-4f));
        ucol(sh, "ColorB", Color.withAlpha(colorB, Math.round(((colorB >>> 24) & 0xFF) * opacity())));

        int r = (ca >>> 16) & 0xFF, g = (ca >>> 8) & 0xFF, b = ca & 0xFF, a = (ca >>> 24) & 0xFF;
        Matrix4f mat = ctx.getMatrices().peek().getPositionMatrix();
        BufferBuilder bb = Tessellator.getInstance().begin(VertexFormat.DrawMode.QUADS, VertexFormats.POSITION_TEXTURE_COLOR);
        bb.vertex(mat, x0, y0, 0f).texture(x0 - cx, y0 - cy).color(r, g, b, a);
        bb.vertex(mat, x0, y1, 0f).texture(x0 - cx, y1 - cy).color(r, g, b, a);
        bb.vertex(mat, x1, y1, 0f).texture(x1 - cx, y1 - cy).color(r, g, b, a);
        bb.vertex(mat, x1, y0, 0f).texture(x1 - cx, y0 - cy).color(r, g, b, a);
        BufferRenderer.drawWithGlobalProgram(bb.end());
        RenderSystem.enableCull();
    }

    private static void u1(ShaderProgram s, String n, float v) { GlUniform u = s.getUniform(n); if (u != null) u.set(v); }
    private static void u2(ShaderProgram s, String n, float a, float b) { GlUniform u = s.getUniform(n); if (u != null) u.set(a, b); }
    private static void ui(ShaderProgram s, String n, int v) { GlUniform u = s.getUniform(n); if (u != null) u.set(v); }
    private static void u4f(ShaderProgram s, String n, float a, float b, float c, float d) { GlUniform u = s.getUniform(n); if (u != null) u.set(a, b, c, d); }
    private static void ucol(ShaderProgram s, String n, int c) { GlUniform u = s.getUniform(n); if (u != null) u.set(Color.rf(c), Color.gf(c), Color.bf(c), Color.af(c)); }
}

        if (reg == null) { reg = ModernAtlas.load("regular"); med = ModernAtlas.load("medium"); semi = ModernAtlas.load("semibold"); }

    private ModernAtlas atlas(Weight w) { ensure(); return w == Weight.REGULAR ? reg : w == Weight.SEMIBOLD ? semi : med; }

    @Override public boolean isResolutionIndependent() { return true; }
    @Override public float width(String s, Weight w, float size) { return atlas(w).metrics.width(s, size); }
    @Override public float ascent(Weight w, float size) { return atlas(w).metrics.ascent(size); }
    @Override public float descent(Weight w, float size) { return atlas(w).metrics.descent(size); }
    @Override public float lineHeight(Weight w, float size) { return atlas(w).metrics.lineHeight(size); }

    @Override public List<String> wrap(String s, Weight w, float size, float maxWidth) {
        MsdfMetrics m = atlas(w).metrics;
        List<String> out = new ArrayList<>();
        StringBuilder line = new StringBuilder();
        for (String word : s.split(" ")) {
            String cand = line.length() == 0 ? word : line + " " + word;
            if (m.width(cand, size) <= maxWidth || line.length() == 0) line = new StringBuilder(cand);
            else { out.add(line.toString()); line = new StringBuilder(word); }
        }
        if (line.length() > 0) out.add(line.toString());
        return out;
    }

    @Override public float draw(String text, float x, float y, TextStyle st) {
        if (st.align == Align.CENTER) x -= width(text, st.weight, st.size) / 2f;
        else if (st.align == Align.RIGHT) x -= width(text, st.weight, st.size);
        // shadow effect = offset dark pass first
        if (st.effect instanceof TextEffect.Shadow sh)
            drawRun(text, x + sh.dx(), y + sh.dy(), st.weight, st.size, sh.color(), 0f, 0, sh.softness(), sh.color());
        float outlineW = st.effect instanceof TextEffect.Outline o ? o.widthPx() : 0f;
        int outlineC = st.effect instanceof TextEffect.Outline o2 ? o2.color() : 0;
        float glowR = st.effect instanceof TextEffect.Glow g ? g.radius() : 0f;
        int glowC = st.effect instanceof TextEffect.Glow g2 ? g2.color() : 0;
        return drawRun(text, x, y, st.weight, st.size, st.color, outlineW, outlineC, glowR, glowC);
    }

    @Override public void drawWrapped(String text, float x, float y, float maxWidth, TextStyle st) {
        float lh = lineHeight(st.weight, st.size);
        float cy = y;
        for (String line : wrap(text, st.weight, st.size, maxWidth)) { draw(line, x, cy, st); cy += lh; }
    }

    private float drawRun(String s, float x, float yTop, Weight w, float size, int color,
                          float outlineW, int outlineC, float glowR, int glowC) {
        ensure();
        if (!UiShaders.ready() || s.isEmpty()) return x;
        ModernAtlas a = atlas(w); a.ensureTexture();
        MsdfMetrics m = a.metrics;
        float baseY = yTop + size * ASCENT;
        int r = (color >>> 16) & 0xFF, g = (color >>> 8) & 0xFF, b = color & 0xFF, al = (color >>> 24) & 0xFF;

        List<float[]> quads = new ArrayList<>();
        float pen = x;
        for (int i = 0; i < s.length(); i++) {
            MsdfMetrics.Glyph gl = m.get(s.charAt(i));
            if (gl == null) gl = m.get((int) '?');
            if (gl == null) continue;
            if (gl.hasBounds)
                quads.add(new float[]{pen + gl.pl * size, baseY - gl.pt * size, pen + gl.pr * size, baseY - gl.pb * size, gl.u0, gl.v0, gl.u1, gl.v1});
            pen += gl.advance * size;
        }
        if (quads.isEmpty()) return pen;

        ShaderProgram sh = UiShaders.TEXT;
        RenderSystem.enableBlend(); RenderSystem.defaultBlendFunc(); RenderSystem.disableCull();
        RenderSystem.setShader(() -> sh);
        RenderSystem.setShaderTexture(0, a.textureId);
        u1(sh, "PxRange", m.distanceRange);
        u1(sh, "OutlineWidth", outlineW); ucol(sh, "OutlineColor", outlineC);
        u1(sh, "GlowRange", glowR); ucol(sh, "GlowColor", glowC);

        Matrix4f mat = ctx.getMatrices().peek().getPositionMatrix();
        BufferBuilder bb = Tessellator.getInstance().begin(VertexFormat.DrawMode.QUADS, VertexFormats.POSITION_TEXTURE_COLOR);
        for (float[] q : quads) {
            bb.vertex(mat, q[0], q[1], 0f).texture(q[4], q[5]).color(r, g, b, al);
            bb.vertex(mat, q[0], q[3], 0f).texture(q[4], q[7]).color(r, g, b, al);
            bb.vertex(mat, q[2], q[3], 0f).texture(q[6], q[7]).color(r, g, b, al);
            bb.vertex(mat, q[2], q[1], 0f).texture(q[6], q[5]).color(r, g, b, al);
        }
        BufferRenderer.drawWithGlobalProgram(bb.end());
        RenderSystem.enableCull();
        return pen;
    }

    private static void u1(ShaderProgram s, String n, float v) { GlUniform u = s.getUniform(n); if (u != null) u.set(v); }
    private static void ucol(ShaderProgram s, String n, int c) { GlUniform u = s.getUniform(n); if (u != null) u.set(Color.rf(c), Color.gf(c), Color.bf(c), Color.af(c)); }
}
